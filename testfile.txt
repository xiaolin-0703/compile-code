int classify(char ch) {
    if (ch >= 'A' && ch <= 'Z') {
        return 1;
    }
    if (ch >= 'a' && ch <= 'z') {
        return 2;
    }
    if (ch >= '0' && ch <= '9') {
        return 3;
    }
    return 4;
}
char toUpper(char ch) {
    int value;
    if (ch >= 'a' && ch <= 'z') {
        value = (int)ch - (int)'a' + (int)'A';
        return (char)value;
    }
    return ch;
}
int main() {
    char input[13];
    char normalized[13];
    int count[5] = {0, 0, 0, 0, 0};
    int i = 0;
    int kind;
    int score = 0;
    while (i < 12) {
        input[i] = (char)get_char();
        normalized[i] = toUpper(input[i]);
        kind = classify(input[i]);
        count[kind] = count[kind] + 1;
        switch (kind) {
        case 1:
            score = score + 4;
            break;
        case 2:
            score = score + 3;
            break;
        case 3:
            score = score + 2;
            break;
        default:
            score = score + 1;
        }
        i = i + 1;
    }
    input[12] = '\0';
    normalized[12] = '\0';
    printf("raw=%s\n", input);
    printf("upper=%s\n", normalized);
    printf("U=%d L=%d D=%d O=%d\n", count[1], count[2], count[3], count[4]);
    printf("score=%d\n", score);
    i = 11;
    printf("reverse=");
    while (i >= 0) {
        printf("%c", normalized[i]);
        i = i - 1;
    }
    printf("\n");
    return 0;
}
